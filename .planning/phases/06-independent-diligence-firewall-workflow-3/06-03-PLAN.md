```python
def synthesize_verdict(findings: list[AnalystSignal], *, liquidity_zero: bool, params: dict) -> Literal["go","no_go"]:
    if liquidity_zero: return "no_go"                       # zero-liquidity hard fail
    money = [s for s in findings if s.dimension in MONEY_TOUCHING]
    if any(not dimension_passed(s, params=params) for s in money): return "no_go"   # ONE money dim fails -> NO-GO
    return "go" if money and all(dimension_passed(s, params=params) for s in money) else "no_go"  # GO needs all money backed; ties -> NO-GO
```
