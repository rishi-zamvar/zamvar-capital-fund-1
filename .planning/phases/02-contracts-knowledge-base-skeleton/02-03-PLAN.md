```python
def insert_claim(conn, node: KBNode, evidence: list[EvidenceSource], embed=None, embed_text=None) -> int:
    if not evidence:
        raise ValueError("cannot store a claim without a CITES->EvidenceSource edge")
    try:
        nid = conn.execute("INSERT INTO kb_node (kind, ref_id, attrs_json, confidence, created_at) VALUES (?,?,?,?,?)",
                           (node.kind, node.ref_id, node.attrs_json, node.confidence, _now())).lastrowid
        for ev in evidence:
            eid = _upsert_evidence(conn, ev)             # kind='evidence_source' node, dedup on url+snippet_hash
            conn.execute("INSERT INTO kb_edge (src_id, rel, dst_id, attrs_json, created_at) VALUES (?,?,?,?,?)",
                         (nid, "CITES", eid, None, _now()))
        if embed is not None and embed_text is not None:
            conn.execute("INSERT INTO kb_vec (node_id, embedding) VALUES (?, ?)",
                         (nid, serialize_float32(embed(embed_text))))
        conn.commit(); return nid
    except Exception:
        conn.rollback(); raise
```
